**What happened**:
When a high-priority workload with a topology spreading constraint attempts to preempt lower-priority workloads, flavor assignment evaluates the target domains as banned by spreading limits *before* the preemption logic is evaluated.

Consequently, preemption never triggers. The high-priority workload fails flavor assignment immediately with `TopologyPlacementFailed`, while lower-priority workloads remain admitted.

**What you expected to happen**:
High-priority workloads should be able to preempt lower-priority workloads. Preemption candidates (lower-priority workloads that could be evicted/preempted) should be excluded from the spreading count/ban calculation so that valid placement domains can be evaluated for preemption.

**How to reproduce it (as minimally and precisely as possible)**:
```bash
CTX=kind-tas-spread
kubectl --context $CTX -n ts1 delete jobs --all --wait=true >/dev/null

# 1. Patch CQ to allow preemption and create priority classes
kubectl --context $CTX patch clusterqueue tas-cq --type=merge -p '{"spec":{"preemption":{"withinClusterQueue":"LowerPriority","reclaimWithinCohort":"Never"}}}'
kubectl --context $CTX apply -f - <<'EOF'
apiVersion: kueue.x-k8s.io/v1beta2
kind: WorkloadPriorityClass
metadata: {name: lowp}
value: 1
---
apiVersion: kueue.x-k8s.io/v1beta2
kind: WorkloadPriorityClass
metadata: {name: highp}
value: 1000
EOF

# 2. Set node capacity to 1 GPU each
for n in tas-spread-worker tas-spread-worker2 tas-spread-worker3 tas-spread-worker4 tas-spread-worker5 tas-spread-worker6 tas-spread-worker7 tas-spread-worker8; do
  kubectl --context $CTX patch node $n --subresource=status --type=merge -p '{"status":{"capacity":{"example.com/gpu":"1"},"allocatable":{"example.com/gpu":"1"}}}' >/dev/null
done
sleep 10

# 3. Fill nodes with 8 low-priority jobs
mklow() { kubectl --context $CTX apply -f - <<EOF
apiVersion: batch/v1
kind: Job
metadata:
  name: $1
  namespace: ts1
  labels: {kueue.x-k8s.io/queue-name: lq, kueue.x-k8s.io/priority-class: lowp, spread-group: gh4}
spec:
  parallelism: 1
  completions: 1
  suspend: true
  template:
    metadata: {annotations: {kueue.x-k8s.io/podset-required-topology: cloud.provider.com/topology-block}}
    spec:
      restartPolicy: Never
      containers:
      - name: c
        image: registry.k8s.io/e2e-test-images/agnhost:2.66.1
        args: [netexec]
        resources: {requests: {example.com/gpu: "1"}, limits: {example.com/gpu: "1"}}
EOF
}
for i in 1 2 3 4 5 6 7 8; do mklow h4-low-$i; sleep 4; done
sleep 10

# 4. Deploy 1 high-priority job with maxShare 1u
kubectl --context $CTX apply -f - <<'EOF'
apiVersion: batch/v1
kind: Job
metadata:
  name: h4-high
  namespace: ts1
  labels: {kueue.x-k8s.io/queue-name: lq, kueue.x-k8s.io/priority-class: highp, spread-group: gh4}

  parallelism: 1
  completions: 1
  suspend: true
  template:
    metadata:
      annotations:
        kueue.x-k8s.io/podset-required-topology: cloud.provider.com/topology-block
        kueue.x-k8s.io/podset-topology-spreading: '{"workloadLabelSelectors":[{"key":"spread-group","operator":"In","values":["gh4"]}],"rules":[{"topologyKey":"cloud.provider.com/topology-block","maxShareAllowingPlacement":"1u"}]}'
    spec:
      restartPolicy: Never
      containers:
      - name: c
        image: registry.k8s.io/e2e-test-images/agnhost:2.66.1

        resources: {requests: {example.com/gpu: "1"}, limits: {example.com/gpu: "1"}}
EOF

# 5. Observe high priority job is False/TopologyPlacementFailed and no preemption occurs
```
