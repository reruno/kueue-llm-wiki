+// Stop suspends the SparkApplication and explicitly tears down its
+// driver + executor pods in an order that avoids two failure modes:
+//
+//  1. Suspend the SparkApplication (spec.suspend=true). Spark Operator
+//     interprets this as a graceful pause and will tear down the driver
+//     so AppState can transition back to Running on resume.
+//     Force-deleting the driver ourselves instead would make Spark
+//     Operator mark AppState=Failed, which the framework Finished()
+//     check turns into Workload.Finished=True — that prematurely ends
+//     the kueue retry cycle.
+//  2. Wait briefly for the driver pod to be gone. The driver's Spark
+//     scheduler can issue "create executor" API calls right up until
+//     its container terminates; if we delete executors before the
+//     driver is dead, those late calls would leave orphan executors
+//     (owner=SparkApplication only, no driver pod ref) that linger
+//     across cycles.
+//  3. Delete all executor pods. With the driver dead they cannot be
+//     respawned mid-cleanup.
+func (j *SparkApplication) Stop(ctx context.Context, c client.Client, podSetsInfo []podset.PodSetInfo, _ jobframework.StopReason, _ string) (bool, error) {
+	if ptr.Deref(j.Spec.Suspend, false) {
+		return false, nil
+	}
+
+	// Step 1: Suspend SparkApplication (graceful — does not trigger Failed).
+	obj := j.Object()
+	if err := clientutil.Patch(ctx, c, obj, func() (bool, error) {
+		j.Suspend()
+		if podSetsInfo != nil {
+			j.RestorePodSetsInfo(podSetsInfo)
+		}
+		return true, nil
+	}); err != nil {
+		return false, err
+	}
+
+	// Step 2: Wait for Spark Operator to actually kill the driver. Bounded
+	// poll: at most 5s. Longer than this we accept the residual race and
+	// move on — the next eviction cycle will clean up any orphan executor.
+	var driverList corev1.PodList
+	if err := c.List(ctx, &driverList,
+		client.InNamespace(j.GetNamespace()),
+		client.MatchingLabels{
+			sparkcommon.LabelSparkAppName: j.GetName(),
+			sparkcommon.LabelSparkRole:    sparkcommon.SparkRoleDriver,
+		},
+	); err == nil && len(driverList.Items) > 0 {

+//     respawned mid-cleanup.

+	if ptr.Deref(j.Spec.Suspend, false) {
+		return false, nil
+	}

+	// Step 1: Suspend SparkApplication (graceful — does not trigger Failed).
+	obj := j.Object()
+	if err := clientutil.Patch(ctx, c, obj, func() (bool, error) {
+		j.Suspend()
+		if podSetsInfo != nil {
+			j.RestorePodSetsInfo(podSetsInfo)
+		}
+		return true, nil
+	}); err != nil {
+		return false, err
+	}

+	// Step 2: Wait for Spark Operator to actually kill the driver. Bounded
+	// poll: at most 5s. Longer than this we accept the residual race and
+	// move on — the next eviction cycle will clean up any orphan executor.
+	var driverList corev1.PodList
+	if err := c.List(ctx, &driverList,
+		client.InNamespace(j.GetNamespace()),
+		client.MatchingLabels{
+			sparkcommon.LabelSparkAppName: j.GetName(),
+			sparkcommon.LabelSparkRole:    sparkcommon.SparkRoleDriver,
+		},
+	); err == nil && len(driverList.Items) > 0 {
+		driverKeys := make([]client.ObjectKey, 0, len(driverList.Items))
+		for i := range driverList.Items {
+			driverKeys = append(driverKeys, client.ObjectKeyFromObject(&driverList.Items[i]))
+		}
+		_ = wait.PollUntilContextTimeout(ctx, 200*time.Millisecond, 5*time.Second, true, func(ctx context.Context) (bool, error) {
+			for _, key := range driverKeys {
+				var p corev1.Pod
+				err := c.Get(ctx, key, &p)
+				if err == nil {
+					return false, nil
+				}
+				if !apierrors.IsNotFound(err) {
+					return false, err
+				}
+			}
+			return true, nil
+		})
