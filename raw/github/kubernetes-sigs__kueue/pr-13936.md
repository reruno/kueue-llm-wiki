## Description

#### What type of PR is this?

/kind feature

#### What this PR does / why we need it:

Adds a MultiKueue control-plane performance benchmark with one manager and three worker `envtest` control planes. It runs the real Kueue core and MultiKueue controllers under production-default controller settings, generates manager-side Workloads, and reports throughput, admission and quota-reservation latency, worker distribution, timing, and watch gaps.

The new `make test-performance-multikueue` target runs the benchmark and checks its result against a strict scenario-specific range. The committed range is provisional: this is the repository-side, non-alerting phase, and it should be recalibrated on the dedicated CI worker before TestGrid alerting is enabled.

#### Which issue(s) this PR fixes:

Part of https://github.com/kubernetes-sigs/kueue/issues/8564

#### Special notes for your reviewer:

- The baseline intentionally retains the deployed remote-client defaults. Its throughput is therefore most sensitive to changes in remote API request volume; it is not a CPU or memory capacity benchmark.
- The external `kubernetes/test-infra` periodic and dedicated-worker calibration remain follow-up work.
- On the refreshed branch, the full 120-Workload target admitted 120/120 at 1.289 Workloads/s, with 90.254s admission P95, 70ms quota-reservation P95, zero watch gaps, and all checker tests passing. Targeted race tests and `go vet` also passed.

AI tools were used to assist with implementation and review. The final diff was manually reviewed and validated.

+func (c *observationCollector) observe(wl *kueue.Workload, observedAt time.Time) error {
+	key := client.ObjectKeyFromObject(wl)
+	timing := c.workloads[key]
+	if timing == nil {
+		timing = &workloadTiming{}
+		c.workloads[key] = timing
+	}
+	if timing.createdAt.IsZero() {
+		// Falling back to CreationTimestamp would silently drop the measurement to the
+		// one-second granularity this annotation exists to avoid, so refuse to guess.
+		value := wl.Annotations[benchmarkCreatedAtAnnotation]
+		if value == "" {
+			return fmt.Errorf("workload %s has no %s annotation", key, benchmarkCreatedAtAnnotation)
+		}
+		parsed, err := time.Parse(time.RFC3339Nano, value)
+		if err != nil {
+			return fmt.Errorf("parse %s annotation of workload %s: %w", benchmarkCreatedAtAnnotation, key, err)
+		}
+		timing.createdAt = parsed
+	}
+	if timing.quotaReservedAt.IsZero() {
+		if condition := apimeta.FindStatusCondition(wl.Status.Conditions, kueue.WorkloadQuotaReserved); condition != nil && condition.Status == metav1.ConditionTrue {
+			timing.quotaReservedAt = observedAt
+		}
+	}
+	if timing.dispatchReadyAt.IsZero() {
+		for i := range wl.Status.AdmissionChecks {
+			check := &wl.Status.AdmissionChecks[i]
+			if check.Name == multiKueueAdmissionCheck && check.State == kueue.CheckStateReady {
+				timing.dispatchReadyAt = observedAt
+				break
+			}
+		}
+	}
+	if timing.admittedAt.IsZero() {
+		if condition := apimeta.FindStatusCondition(wl.Status.Conditions, kueue.WorkloadAdmitted); condition != nil && condition.Status == metav1.ConditionTrue {
+			timing.admittedAt = observedAt
+			c.admitted++
+		}
+	}
+	if timing.cluster == "" && wl.Status.ClusterName != nil {
+		timing.cluster = *wl.Status.ClusterName
+	}
+	return nil

+func (c *observationCollector) observe(wl *kueue.Workload, observedAt time.Time) error {
+	key := client.ObjectKeyFromObject(wl)
+	timing := c.workloads[key]
+	if timing == nil {
+		timing = &workloadTiming{}
+		c.workloads[key] = timing
+	}
+	if timing.createdAt.IsZero() {
+		// Falling back to CreationTimestamp would silently drop the measurement to the
+		// one-second granularity this annotation exists to avoid, so refuse to guess.
+		value := wl.Annotations[benchmarkCreatedAtAnnotation]
+		if value == "" {
+			return fmt.Errorf("workload %s has no %s annotation", key, benchmarkCreatedAtAnnotation)
+		}
+		parsed, err := time.Parse(time.RFC3339Nano, value)
+		if err != nil {
+			return fmt.Errorf("parse %s annotation of workload %s: %w", benchmarkCreatedAtAnnotation, key, err)
+		}
+		timing.createdAt = parsed
+	}
+	if timing.quotaReservedAt.IsZero() {
+		if condition := apimeta.FindStatusCondition(wl.Status.Conditions, kueue.WorkloadQuotaReserved); condition != nil && condition.Status == metav1.ConditionTrue {
+			timing.quotaReservedAt = observedAt
+		}
+	}
+	if timing.dispatchReadyAt.IsZero() {
+		for i := range wl.Status.AdmissionChecks {
+			check := &wl.Status.AdmissionChecks[i]

+				timing.dispatchReadyAt = observedAt
+				break
+			}
+		}
+	}
+	if timing.admittedAt.IsZero() {
+		if condition := apimeta.FindStatusCondition(wl.Status.Conditions, kueue.WorkloadAdmitted); condition != nil && condition.Status == metav1.ConditionTrue {
+			timing.admittedAt = observedAt

+		}
+	}
+	if timing.cluster == "" && wl.Status.ClusterName != nil {
+		timing.cluster = *wl.Status.ClusterName
+	}
