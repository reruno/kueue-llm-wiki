```mermaid

flowchart TD
    CheckTAS{"TopologyAwareScheduling"}

    CheckTAS -- Enabled --> CheckSchedLib{"SchedulerLibraryIntegration"}
    
    CheckSchedLib -- Disabled --> TASInternal["tas-internal"]
    CheckSchedLib -- Enabled --> CheckDeep{"SchedulerLibraryDeepIntegration"}
    
    CheckDeep -- Disabled --> TASFeas["tas-schedlib-feasibility"]
    CheckDeep -- Enabled --> TASDeep["tas-schedlib-deep"]

```
